{{/* layouts/partials/breadcrumb-ld.html */}} {{/* Universal JSON-LD BreadcrumbList. Works on Home, Section pages, blog posts, service pages, etc. */}} {{/* 1) Build a slice of crumbs, always starting with “Home” */}} {{ $crumbs := slice (dict "name" "Home" "url" (absURL "/")) }} {{/* 2) If this is a section (e.g. /blog/, /services/), append it as a crumb */}} {{ if .IsSection }} {{ $crumbs = $crumbs | append (dict "name" (title .Section) "url" .Permalink) }} {{/* 3) If this is a single page (any “Kind=page”), walk its .Ancestors */}} {{ else if .IsPage }} {{ range .Ancestors }} {{ $crumbs = $crumbs | append (dict "name" .Title "url" .Permalink) }} {{ end }} {{ $crumbs = $crumbs | append (dict "name" .Title "url" .Permalink) }} {{/* 4) If this is the homepage, we leave $crumbs as just “Home” */}} {{ end }}
