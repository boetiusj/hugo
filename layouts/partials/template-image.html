{{/* ------------------------------------------------------------- Template: template-image.html -------------------------------------------------------------- */}} {{- $eager := or (eq .eager true) (eq .eager "true") -}} {{/* 1. Source ------------------------------------------------- */}} {{- $src := or .Destination .src -}} {{- if not $src }} {{ errorf "Image Optimization: the partial requires either 'src' or 'Destination'. Got: %#v" . }} {{- end }} {{- $image := $src }} {{- $eager := eq .eager true }} {{- $thumb := .thumb | default false }} {{/* 2. Locate the resource (bundle first, then /assets) -------- */}} {{- $resource := .Page.Resources.GetMatch $image -}} {{- if not $resource }} {{ $resource = resources.Get (strings.TrimPrefix "/" $image) }} {{- end }} {{- if not $resource }} {{ errorf "Image not found: %s" $image }} {{- end }} {{/* 3. Optional thumbnail crop -------------------------------- */}} {{- if $thumb }} {{ $resource = $resource.Fill "200x160 smart" }} {{- end }} {{/* 4. Alt text ---------------------------------------------- */}} {{- $baseAlt := replaceRE "^.*/([^/]+)\\.[^.]+$" "$1" $image -}} {{- $alt := .alt | default $baseAlt -}} {{/* 5. Sizes & output ---------------------------------------- */}} {{- $width := $resource.Width | int -}} {{- $height := $resource.Height | int -}} {{- if ge $width 500 }} {{- $webp := $resource.Fit (printf "%dx%d q75 webp" $width $height) -}} {{- $avif := $resource.Fit (printf "%dx%d q50 avif" $width $height) -}} {{- /* Build srcset entries only up to the image’s natural width */ -}} {{- $entries := slice -}} {{- if ge $width 1280 -}} {{- $entries = $entries | append (printf "%s 1280w" ($resource.Resize "1280x webp q75").Permalink) -}} {{- end -}} {{- if ge $width 960 -}} {{- $entries = $entries | append (printf "%s 960w" ($resource.Resize "960x webp q75").Permalink) -}} {{- end -}} {{- if ge $width 640 -}} {{- $entries = $entries | append (printf "%s 640w" ($resource.Resize "640x webp q75").Permalink) -}} {{- end -}} {{- if ge $width 320 -}} {{- $entries = $entries | append (printf "%s 320w" ($resource.Resize "320x webp q75").Permalink) -}} {{- end -}} {{- $srcset := delimit $entries ", " -}} {{- else }} {{- end }}
