{{/* ----------------------------------------------- Shortcode: figure1st 1) Grab src= or page.Params.image, default to "" 2) Trim all “/” so "/foo/" → "foo", "///" → "" 3) Strip any accidental leading "images/" 4) If empty, bail out early 5) Otherwise try assets/images/...; if missing, fall back to /images/... ----------------------------------------------- */}} {{- /* 1. raw input, never nil */ -}} {{- $raw := .Get "src" | default .Page.Params.image | default "" -}} {{- /* 2. strip any leading/trailing slashes */ -}} {{- $trim := trim $raw "/" -}} {{- /* 3. remove accidental "images/" prefix */ -}} {{- $trim = strings.TrimPrefix $trim "images/" -}} {{- /* 4. nothing to do? exit */ -}} {{- if eq $trim "" -}} {{ warnf "figure1st: no src defined in %s" .Page.File.Path }} {{- return -}} {{- end -}} {{- /* 5. try to pipeline from assets/images */ -}} {{- $asset := printf "images/%s" $trim -}} {{- $res := resources.Get $asset -}} {{- if and $res (eq $res.MediaType.Type "image") -}} {{- else -}} {{ warnf "figure1st: falling back to static /images/%s" $trim }} {{- end -}}
